--------------------------------------------------------------------------------
TIER 1 — base aggregates          HOME TABLE: fact_placement
--------------------------------------------------------------------------------

Total Spend = SUM ( fact_placement[spend_to_date] )
-- expect 5,408,750.68   population: all 595 placements

Contracted Impressions = SUM ( fact_placement[contracted_impressions] )
-- expect 518,781,947   population: all 595, FULL flight (not the to-date denominator)

Contracted To Date = SUM ( fact_placement[contracted_to_date] )
-- expect 427,728,324   population: all 595, prorated to elapsed days

Verified Impressions = SUM ( fact_placement[verified_to_date] )
-- expect 418,677,830   population: all 595

Gross Shortfall = SUM ( fact_placement[shortfall] )
-- expect 10,532,493   population: the 227 behind. Column is floored at 0, so no
-- conditional is needed and the measure is verifiable without reading DAX.

Over-delivery Offset = SUM ( fact_placement[over_delivery] )
-- expect 1,481,999   population: the 367 ahead

Placements = COUNTROWS ( fact_placement )
-- expect 595

Placements Behind =
CALCULATE ( COUNTROWS ( fact_placement ), fact_placement[shortfall] > 0 )
-- expect 227   behind by ANY amount

Flagged Placements =
CALCULATE ( COUNTROWS ( fact_placement ), fact_placement[is_flagged] = TRUE() )
-- expect 82   past the -5% attention line

Live Issues =
CALCULATE ( COUNTROWS ( fact_placement ), fact_placement[status] = "live_issue" )
-- expect 21

Completed Shortfalls =
CALCULATE ( COUNTROWS ( fact_placement ), fact_placement[status] = "completed_shortfall" )
-- expect 61

Campaigns Affected =
CALCULATE ( DISTINCTCOUNT ( fact_placement[campaign_id] ), fact_placement[is_flagged] = TRUE() )
-- expect 8

Markets Affected =
CALCULATE ( DISTINCTCOUNT ( dim_site[city] ), fact_placement[is_flagged] = TRUE() )
-- expect 6

Total Negative Delivery Exposure = SUM ( fact_placement[billed_shortfall] )
-- expect 134,655.35   population: the 227 behind
-- *** NOT the same measure as Flagged Media-Value Exposure. 17,598 apart. ***

Flagged Media-Value Exposure =
CALCULATE ( SUM ( fact_placement[billed_shortfall] ), fact_placement[is_flagged] = TRUE() )
-- expect 117,057.70   population: the 82 flagged
-- *** If this equals the measure above, the filter is missing. ***

Preventable Exposure =
CALCULATE ( SUM ( fact_placement[preventable_exposure] ), fact_placement[status] = "live_issue" )
-- expect 36,254.56   MODELLED. Never add to either exposure measure above:
-- one is money already spent, this has not happened yet.

--------------------------------------------------------------------------------
TIER 2 — derived                  HOME TABLE: fact_placement
--------------------------------------------------------------------------------

Net Shortfall = [Gross Shortfall] - [Over-delivery Offset]
-- expect 9,050,494

Masked Shortfall % = DIVIDE ( [Over-delivery Offset], [Gross Shortfall] )
-- expect 14.07%   format as percentage, 0 dp

Delivery Variance % =
DIVIDE ( [Verified Impressions] - [Contracted To Date], [Contracted To Date] )
-- expect -2.12%   format as percentage, 1 dp

Blended CPM = DIVIDE ( [Total Spend], [Verified Impressions] ) * 1000
-- expect 12.92   NEVER split this by media_type: digital and static CPM are not
-- comparable, because the model does not represent digital share-of-loop.

--------------------------------------------------------------------------------
TIER 3 — detection                HOME TABLE: fact_detection_events
--------------------------------------------------------------------------------

Detection Events = COUNTROWS ( fact_detection_events )
-- expect 102   every detected fault

Alerted Events =
CALCULATE (
    COUNTROWS ( fact_detection_events ),
    NOT ISBLANK ( fact_detection_events[detection_delay_days] )
)
-- expect 90   detected AND crossed the alert line

Faults Recovered =
CALCULATE ( COUNTROWS ( fact_detection_events ), fact_detection_events[recovered] = TRUE() )
-- expect 32

Median Detection Delay =
MEDIANX (
    FILTER ( fact_detection_events, NOT ISBLANK ( fact_detection_events[detection_delay_days] ) ),
    fact_detection_events[detection_delay_days]
)
-- expect 4   population: alerted events (90)

Median Reconciliation Delay =
MEDIANX (
    FILTER ( fact_detection_events, NOT ISBLANK ( fact_detection_events[detection_delay_days] ) ),
    fact_detection_events[reconciliation_delay_days]
)
-- expect 31   population: alerted events (90)
-- Pair 4-against-31 on one card row. That contrast is the project's headline.

Median Days Remaining at Detection =
MEDIANX (
    FILTER (
        fact_detection_events,
        NOT ISBLANK ( fact_detection_events[detection_delay_days] )
            && RELATED ( fact_placement[in_flight] ) = TRUE()
    ),
    fact_detection_events[days_remaining_at_detection]
)
-- expect 35   population: alerted AND still in flight (23)
-- *** BOTH filters required. Alerted-only gives 26. ***

Avg Detection Delay =
AVERAGEX (
    FILTER ( fact_detection_events, NOT ISBLANK ( fact_detection_events[detection_delay_days] ) ),
    fact_detection_events[detection_delay_days]
)
-- expect 4.82   prefer the median on cards; the mean is pulled by a long tail

--------------------------------------------------------------------------------
TIER 4 — cumulative lines         HOME TABLE: fact_delivery
--------------------------------------------------------------------------------

Daily Verified = SUM ( fact_delivery[verified_impressions] )

Cumulative Verified =
CALCULATE (
    [Daily Verified],
    FILTER ( ALLSELECTED ( dim_date[date] ), dim_date[date] <= MAX ( dim_date[date] ) )
)

Cumulative Contracted =
VAR DaysSoFar =
    CALCULATE (
        DISTINCTCOUNT ( fact_delivery[date] ),
        FILTER ( ALLSELECTED ( dim_date[date] ), dim_date[date] <= MAX ( dim_date[date] ) )
    )
VAR PlannedPerDay =
    SUMX ( VALUES ( fact_placement[placement_id] ), fact_placement[planned_daily] )
RETURN
    DaysSoFar * PlannedPerDay
-- Assumes the page is FILTERED TO ONE CAMPAIGN. Every placement in a campaign
-- shares its flight window, which is what makes days x planned/day valid.
-- With "Signal Everywhere" selected the final points are:
--   Cumulative Verified    59,392,728
--   Cumulative Contracted  61,151,454
--   gap                     1,758,726
-- If the lines look wrong, the campaign filter is missing.

--------------------------------------------------------------------------------
TIER 5 — helper tables            Ribbon: New table (NOT a measure)
--------------------------------------------------------------------------------

Bridge =
DATATABLE (
    "Step", STRING, "Order", INTEGER,
    { { "Gross shortfall", 1 }, { "Offset by over-delivery", 2 }, { "Net shortfall", 3 } }
)
-- then sort column Step by column Order

Bridge Value =
SWITCH (
    SELECTEDVALUE ( Bridge[Order] ),
    1, [Gross Shortfall],
    2, -[Over-delivery Offset],
    3, [Net Shortfall]
)
-- measure, home table: Bridge. Feeds the waterfall on page 1.

Analysis Measure =
{
    ( "Flagged media-value exposure", NAMEOF ( [Flagged Media-Value Exposure] ), 0 ),
    ( "Gross shortfall",              NAMEOF ( [Gross Shortfall] ),              1 ),
    ( "Placements behind",            NAMEOF ( [Placements Behind] ),            2 ),
    ( "Verified impressions",         NAMEOF ( [Verified Impressions] ),         3 )
}
-- Field parameter. Web modeling has no "New parameter" button, but a field
-- parameter IS just this calculated table, so it is authored by hand.
-- After creating: rename Value1 -> Measure, Value2 -> Measure Fields,
-- Value3 -> Measure Order; sort Measure by Measure Order; mark Measure Fields
-- as a field-parameter column in Properties.
-- NOTE: a field parameter cannot be the linked field of a drill-through or
-- tooltip page. Use the underlying measure there.
